cmake_minimum_required(VERSION 3.26)
project(CetRuntimeChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(ESSENTIALS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(cet_src "${ESSENTIALS_ROOT}/sources/cyber-engine-tweaks/src")
set(dep "${ESSENTIALS_ROOT}/build/deps")
if(NOT CET_NATIVE_BUILD OR NOT CET_CONTINUATION_DIR)
  message(FATAL_ERROR "Set CET_NATIVE_BUILD to the built root CMake tree and CET_CONTINUATION_DIR to generated method extraction")
endif()
add_executable(cet-lua lua.cpp "${cet_src}/lsqlite3/lsqlite3.cpp")
target_include_directories(cet-lua PRIVATE "${cet_src}" "${CET_CONTINUATION_DIR}")
target_include_directories(cet-lua SYSTEM PRIVATE "${dep}/cet-sol2/include" "${dep}/cet-luajit/src" "${dep}/cet-sqlite")
target_compile_definitions(cet-lua PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN SOL_ALL_SAFETIES_ON SOL_LUAJIT=1 SOL_EXCEPTIONS_SAFE_PROPAGATION=0)
target_compile_options(cet-lua PRIVATE /EHsc /utf-8 /UNDEBUG)
target_link_libraries(cet-lua PRIVATE "${CET_NATIVE_BUILD}/cet_luajit.lib" "${CET_NATIVE_BUILD}/cet_sqlite.lib")
add_executable(cet-lifecycle lifecycle.cpp "${cet_src}/common/Runtime.cpp")
target_include_directories(cet-lifecycle PRIVATE "${cet_src}")
target_include_directories(cet-lifecycle SYSTEM PRIVATE "${dep}/minhook/include")
target_compile_definitions(cet-lifecycle PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(cet-lifecycle PRIVATE /EHsc /UNDEBUG)
add_executable(cet-graphics graphics.cpp "${cet_src}/common/D3DUpload.cpp")
target_include_directories(cet-graphics PRIVATE "${cet_src}")
target_compile_definitions(cet-graphics PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(cet-graphics PRIVATE /EHsc /UNDEBUG)
add_executable(cet-config config.cpp "${cet_src}/common/JsonConfig.cpp")
target_include_directories(cet-config PRIVATE "${cet_src}")
target_include_directories(cet-config SYSTEM PRIVATE "${dep}/cet-json/include" "${dep}/spdlog/include")
target_compile_definitions(cet-config PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(cet-config PRIVATE /EHsc /UNDEBUG)
