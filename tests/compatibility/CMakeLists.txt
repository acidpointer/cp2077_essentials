cmake_minimum_required(VERSION 3.26)
project(LegacyCompatibilityCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
get_filename_component(ESSENTIALS_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
include("${ESSENTIALS_ROOT}/cmake/Compatibility.cmake")
find_package(Threads REQUIRED)
enable_testing()
add_executable(compat_core_tests core_tests.cpp)
target_link_libraries(compat_core_tests PRIVATE compat_core Threads::Threads)
if(MSVC)
  target_compile_options(compat_core_tests PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(compat_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME compat_core COMMAND compat_core_tests)
add_executable(compat_live_tests live_tests.cpp)
target_link_libraries(compat_live_tests PRIVATE compat_core Threads::Threads)
if(MSVC)
  target_compile_options(compat_live_tests PRIVATE /W4 /WX /EHa)
else()
  target_compile_options(compat_live_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME compat_live COMMAND compat_live_tests)
set_tests_properties(compat_live PROPERTIES TIMEOUT 45)

if(WIN32)
  add_executable(compat_function_tests function_tests.cpp)
  target_link_libraries(compat_function_tests PRIVATE compat_core)
  target_compile_options(compat_function_tests PRIVATE /W4 /WX /EHsc)
  add_test(NAME compat_function COMMAND compat_function_tests)
  foreach(mode throw persistent detach-failure)
    add_test(NAME compat_function_${mode} COMMAND compat_function_tests ${mode})
    set_tests_properties(compat_function_${mode} PROPERTIES TIMEOUT 45)
  endforeach()
  set_tests_properties(compat_function PROPERTIES TIMEOUT 45)
  set(COMPAT_LEGACY_INCLUDE "" CACHE PATH "Historical SDK include directory")
  set(COMPAT_LEGACY_VFS "" CACHE FILEPATH "Optional Clang case-insensitive historical include overlay")
  if(NOT EXISTS "${COMPAT_LEGACY_INCLUDE}/RED4ext/Callback.hpp")
    message(FATAL_ERROR "Supply COMPAT_LEGACY_INCLUDE; use scripts/test-compatibility.py to extract the recorded SDK read-only")
  endif()
  add_library(legacy_guest SHARED fixtures/legacy_guest.cpp)
  add_library(legacy_guest_unoptimized SHARED fixtures/legacy_guest.cpp)
  target_compile_options(legacy_guest_unoptimized PRIVATE /Od /Ob0)
  add_library(array_guest SHARED fixtures/array_guest.cpp)
  add_library(array_guest_unoptimized SHARED fixtures/array_guest.cpp)
  target_compile_options(array_guest_unoptimized PRIVATE /Od /Ob0)
  add_library(legacy_layout OBJECT fixtures/layout_contract.cpp)
  target_compile_definitions(legacy_layout PRIVATE COMPAT_OLD_LAYOUT=1)
  target_link_libraries(legacy_layout PRIVATE compat_core)
  foreach(target legacy_guest legacy_guest_unoptimized array_guest array_guest_unoptimized legacy_layout)
    target_include_directories(${target} SYSTEM PRIVATE "${COMPAT_LEGACY_INCLUDE}")
    if(COMPAT_LEGACY_VFS)
      target_compile_options(${target} PRIVATE "/clang:-ivfsoverlay" "/clang:${COMPAT_LEGACY_VFS}")
    endif()
    target_compile_options(${target} PRIVATE /EHa /Gy)
    if(NOT target STREQUAL "legacy_layout")
      target_link_options(${target} PRIVATE /OPT:REF)
    endif()
  endforeach()
  add_library(native_layout OBJECT fixtures/layout_contract.cpp)
  target_link_libraries(native_layout PRIVATE compat_core)
  target_include_directories(native_layout SYSTEM PRIVATE "${ESSENTIALS_ROOT}/sources/red4ext-sdk/include")
  add_executable(compat_array_tests array_tests.cpp)
  target_link_libraries(compat_array_tests PRIVATE compat_core)
  target_compile_options(compat_array_tests PRIVATE /W4 /WX /EHa)
  add_dependencies(compat_array_tests array_guest array_guest_unoptimized)
  add_test(NAME compat_array COMMAND compat_array_tests)
  add_test(NAME compat_array_unoptimized COMMAND compat_array_tests unoptimized)
  set_tests_properties(compat_array compat_array_unoptimized PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/bin" TIMEOUT 45)
  add_executable(compat_sdk_tests sdk_tests.cpp)
  target_link_libraries(compat_sdk_tests PRIVATE compat_core)
  target_include_directories(compat_sdk_tests SYSTEM PRIVATE "${ESSENTIALS_ROOT}/sources/red4ext-sdk/include")
  target_compile_definitions(compat_sdk_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(compat_sdk_tests PRIVATE /EHa /Gy)
  target_link_options(compat_sdk_tests PRIVATE /OPT:REF)
  add_dependencies(compat_sdk_tests legacy_guest legacy_guest_unoptimized legacy_layout native_layout)
  add_test(NAME compat_sdk COMMAND compat_sdk_tests)
  add_test(NAME compat_sdk_unoptimized COMMAND compat_sdk_tests unoptimized)
  set_tests_properties(compat_sdk compat_sdk_unoptimized PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

  add_library(dependency SHARED fixtures/init_dll.cpp)
  target_compile_definitions(dependency PRIVATE MODULE_ID=1)
  add_library(initializer SHARED fixtures/init_dll.cpp)
  target_compile_definitions(initializer PRIVATE MODULE_ID=2)
  target_link_libraries(initializer PRIVATE dependency)
  add_executable(compat_platform_tests fixtures/platform.cpp)
  target_compile_options(compat_platform_tests PRIVATE /EHa /Oi)
  add_dependencies(compat_platform_tests initializer)
  add_test(NAME compat_platform COMMAND compat_platform_tests)
  set_tests_properties(compat_platform PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
set_tests_properties(compat_core PROPERTIES TIMEOUT 45)
if(WIN32)
  set_tests_properties(compat_sdk compat_sdk_unoptimized compat_platform PROPERTIES TIMEOUT 45)
endif()
